	\begin{frame}
    \frametitle{Zadání}

		\begin{enumerate}
			\item Program spustí světelný efekt na LED přípravku,
			\vspace{3mm}
			\item LED budou rozsvěceny dle pevného zadání v paměti,
			\vspace{3mm}
			\item pro časování programu bude použita již napsaná funkce \textbf{wait}.
		\end{enumerate}
		
	\end{frame}
